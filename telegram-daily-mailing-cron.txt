# Пример задач cron для рассылок Alltrack.
# ВАЖНО: этот файл сам по себе ничего не запускает.
# Команды нужно добавить в crontab на сервере (crontab -e) от пользователя, у которого есть доступ к проекту.

# Рекомендуемая зона времени cron (иначе может сдвигаться время рассылки)
CRON_TZ=Europe/Moscow

# Штрафы "Без фото" — 1 раз в день в 14:30.
30 14 * * * cd /var/www/www-root/data/www/bimmax.pro/js/alltrack && /usr/bin/php save.php --run-daily-no-photo-fines >> /var/www/www-root/data/www/bimmax.pro/js/alltrack/telegram-daily-no-photo-fines-cron.log 2>&1

# Рассылка "Ответы на перемещения" — проверка каждую минуту.
# Так рассылка уходит независимо от активности пользователей в приложении.
* * * * * cd /var/www/www-root/data/www/bimmax.pro/js/alltrack && /usr/bin/php save.php --run-move-replies-mailing >> /var/www/www-root/data/www/bimmax.pro/js/alltrack/telegram-move-replies-mailing-cron.log 2>&1
